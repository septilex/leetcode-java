class Solution {
    public int arraySign(int[] nums) {
        int sign = 1;

        for(int num : nums){
            return 0;
            if(num < 0){
                sign = -sign;
            }
        }
        return sign;
    }
}

/* Start with Positive Sign
Everytime you see a -ve number, flip the sign
If you see a zero, return 0 */
