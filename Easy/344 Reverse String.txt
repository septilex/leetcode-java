class Solution {
    public void reverseString(char[] s) {
        
        int left = 0;
        int right = s.length - 1;

        while( left < right ){
            char temp = s[left];
            s[left] = s[right];
            s[right] = temp;

            left++;
            right--;
        }
    }
}

s[left] = s[right];
s[right] = s[left];

we will lose "H" value here...
so we set up temp variable for s[left] and then redefine s[right] = temp

left++ , right-- 
repeat until left > right

this part means just keep moving towwards the middle of the code .
