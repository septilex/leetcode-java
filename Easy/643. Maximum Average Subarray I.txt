class Solution {
    public double findMaxAverage(int[] nums, int k) {
        //Create variable sum
        int sum = 0;

        //Build the first window of size k 
        for(int i = 0; i < k ; i++){
            sum += nums[i];
        }

        //Initially, first window sum is always the maximum
        int maxSum = sum;

        //Start your first sliding on the window
        for(int i = k; i < nums.length ; i++){
            //subtract previous element and add the next element
            sum = sum - nums[i-k] + nums[i];
            //Compare the current amx and previous max
            maxSum = Math.max(maxSum, sum);
        }

        return (double) maxSum / k; 
    }
}
