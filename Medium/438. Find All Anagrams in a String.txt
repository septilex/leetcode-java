class Solution {
    public List<Integer> findAnagrams(String s, String p) {

        // Create a new ArrayList to hold answer indices
        List<Integer> result = new ArrayList<>();

        // If pattern is bigger than string
        if(s.length() < p.length()) {
            return result;
        }

        // Count frequencies of pattern
        int[] pCount = new int[26];

        // Count frequencies of current window
        int[] sCount = new int[26];

        // Build first window
        for(int i = 0; i < p.length(); i++) {

            // Count letters in pattern
            pCount[p.charAt(i) - 'a']++;

            // Count letters in first window
            sCount[s.charAt(i) - 'a']++;
        }

        // Check first window
        if(Arrays.equals(pCount, sCount)) {
            result.add(0);
        }

        // Slide the window
        for(int i = p.length(); i < s.length(); i++) {

            // Add new character entering window
            sCount[s.charAt(i) - 'a']++;

            // Remove old character leaving window
            sCount[s.charAt(i - p.length()) - 'a']--;

            // If counts match, save starting index
            if(Arrays.equals(pCount, sCount)) {
                result.add(i - p.length() + 1);
            }
        }

        return result;
    }
}
