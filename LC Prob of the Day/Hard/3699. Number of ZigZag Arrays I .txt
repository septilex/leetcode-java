class Solution {
    static final int MOD = 1_000_000_007;

    public int zigZagArrays(int n, int l, int r) {
        int m = r - l + 1;

        if (n == 1) {
            return m;
        }

        long[][] up = new long[n + 1][m];
        long[][] down = new long[n + 1][m];

        for (int i = 0; i < m; i++) {
            up[1][i] = 1;
            down[1][i] = 1;
        }

        for (int len = 2; len <= n; len++) {

            long[] prefixUp = new long[m + 1];
            long[] prefixDown = new long[m + 1];

            for (int i = 0; i < m; i++) {
                prefixUp[i + 1] = (prefixUp[i] + up[len - 1][i]) % MOD;
                prefixDown[i + 1] = (prefixDown[i] + down[len - 1][i]) % MOD;
            }

            for (int v = 0; v < m; v++) {

                long smaller = prefixDown[v];

                long greater =
                        (prefixUp[m] - prefixUp[v + 1] + MOD) % MOD;

                up[len][v] = smaller;

                down[len][v] = greater;
            }
        }

        long ans = 0;

        for (int i = 0; i < m; i++) {
            ans = (ans + up[n][i]) % MOD;
            ans = (ans + down[n][i]) % MOD;
        }

        return (int) ans;
    }
}
