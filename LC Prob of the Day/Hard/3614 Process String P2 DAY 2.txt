class Solution {

    public char processStr(String s, long k) {

        // Number of characters in input string
        int n = s.length();

        // Store how long the result string is after every step
        long[] len = new long[n + 1];

        // Go through every character
        for (int i = 0; i < n; i++) {

            char ch = s.charAt(i);

            // If we add a letter, string becomes 1 character longer
            if (ch >= 'a' && ch <= 'z') {
                len[i + 1] = len[i] + 1;
            }

            // If we delete a character, length decreases by 1
            else if (ch == '*') {
                len[i + 1] = Math.max(0, len[i] - 1);
            }

            // If we duplicate the string, length doubles
            else if (ch == '#') {
                len[i + 1] = len[i] * 2;
            }

            // Reverse does not change length
            else if (ch == '%') {
                len[i + 1] = len[i];
            }
        }

        // If k is bigger than final string length
        // answer does not exist
        if (k >= len[n]) {
            return '.';
        }

        // Walk backwards through operations
        for (int i = n - 1; i >= 0; i--) {

            char ch = s.charAt(i);

            // Found a normal letter
            if (ch >= 'a' && ch <= 'z') {

                // If k points to this letter
                // this is our answer
                if (k == len[i]) {
                    return ch;
                }
            }

            // Ignore *
            // nothing changes for k
            else if (ch == '*') {

            }

            // Undo duplication
            else if (ch == '#') {

                // Bring k back into first copy
                k %= len[i];
            }

            // Undo reverse
            else if (ch == '%') {

                // Find where this index came from
                k = len[i] - 1 - k;
            }
        }

        return '.';
    }
}
